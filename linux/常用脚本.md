### debian静态 ip 配置

```bash
# /etc/network/interfaces
# The loopback network interface
auto lo
iface lo inet loopback

# The primary network interface
allow-hotplug ens34
#iface ens34 inet dhcp
iface ens34 inet static
    address 10.10.66.3
    netmask 255.255.255.0
    gateway 10.10.66.1
    dns-nameservers 10.10.66.1
```

重启网络

```bash
sudo systemctl restart networking
```

### jenkins 安装脚本

```bash
#!/bin/bash

### CONFIGURATION ###
JENKINS_WAR="/opt/jenkins/jenkins.war"
JENKINS_HOME="/opt/jenkins/jenkins_home"
JAVA_BIN="/usr/local/src/jdk-21.0.9/bin/java"
JVM_OPTS="-Xms1G -Xmx2G"
HTTP_PORT=8080

LOG_DIR="/var/log/jenkins"
PID_FILE="/var/run/jenkins.pid"

mkdir -p "$LOG_DIR"
mkdir -p "$(dirname "$PID_FILE")"

### FUNCTIONS ###

start_jenkins() {
    if [ -f "$PID_FILE" ] && kill -0 $(cat "$PID_FILE") 2>/dev/null; then
        echo "Jenkins is already running (PID: $(cat "$PID_FILE"))."
        exit 1
    fi

    echo "Starting Jenkins..."
    nohup $JAVA_BIN $JVM_OPTS -DJENKINS_HOME="$JENKINS_HOME" \
        -jar "$JENKINS_WAR" --httpPort="$HTTP_PORT" \
        > "$LOG_DIR/jenkins.out" 2>&1 &

    echo $! > "$PID_FILE"
    echo "Jenkins started with PID $(cat "$PID_FILE"). Logs: $LOG_DIR/jenkins.out"
}

stop_jenkins() {
    if [ ! -f "$PID_FILE" ] || ! kill -0 $(cat "$PID_FILE") 2>/dev/null; then
        echo "Jenkins is not running."
        exit 1
    fi

    echo "Stopping Jenkins (PID: $(cat "$PID_FILE"))..."
    kill $(cat "$PID_FILE")
    rm -f "$PID_FILE"
    echo "Jenkins stopped."
}

status_jenkins() {
    if [ -f "$PID_FILE" ] && kill -0 $(cat "$PID_FILE") 2>/dev/null; then
        echo "Jenkins is running (PID: $(cat "$PID_FILE"))."
    else
        echo "Jenkins is not running."
    fi
}

### MAIN LOGIC ###
case "$1" in
    start)
        start_jenkins
        ;;
    stop)
        stop_jenkins
        ;;
    restart)
        stop_jenkins
        sleep 2
        start_jenkins
        ;;
    status)
        status_jenkins
        ;;
    *)
        echo "Usage: $0 {start|stop|restart|status}"
        exit 1
esac
```
